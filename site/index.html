# Clone, fetch local ONNX voices (35 MB TR + 96 MB EN), and build
git clone https://github.com/cengizhankose/tellal
cd tellal
./scripts/fetch-models.sh all
cargo build --release --workspace

# Run live with T3 Code — or try scripted demo mode without T3:
TELLAL_DEMO=demo/fixture.json target/release/tellal
zsh · Loopback 127.0.0.1:47870 → Private Tailnet HTTPS 
# 1. Publish Tellal's loopback mobile server on your tailnet (once)
tailscale serve --bg --https=8455 http://127.0.0.1:47870

# 2. Optional for push-to-talk from phone mic → Wispr Flow
brew install --cask blackhole-2ch

Open the metrics drawer in Tellal → Mobile → click your https://<mac>.<tailnet>.ts.net:8455/ link to copy it. Requests without your Tailscale-User-Login are rejected with 403.
